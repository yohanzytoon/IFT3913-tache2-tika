package org.apache.tika.detect;

import org.apache.tika.detect.CharsetSupersets;
import java.nio.charset.Charset;
import java.util.HashMap;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.junit.jupiter.MockitoExtension;
import org.mockito.*;
import org.junit.jupiter.api.*;
import static org.mockito.Mockito.*;
import static org.junit.jupiter.api.Assertions.*;
import java.util.Collections;
import java.util.Map;

@ExtendWith(MockitoExtension.class)
public class CharsetSupersets_decodeAs_1_0_Test {

    @Mock
    private static Map<String, String> mockSUPERSET_MAP;

    @InjectMocks
    private static CharsetSupersets charsetSupersets;

    @BeforeEach
    public void setUp() {
        HashMap<String, String> superSetMap = new HashMap<>();
        superSetMap.put("UTF-8", "ISO-8859-1");
        when(mockSUPERSET_MAP.get(anyString())).thenReturn("ISO-8859-1");
    }

    @Test
    public void testDecodeAs_withSuperset() {
        Charset detected = Charset.forName("UTF-8");
        Charset result = charsetSupersets.decodeAs(detected);
        assertEquals(Charset.forName("ISO-8859-1"), result);
    }

    @Test
    public void testDecodeAs_withoutSuperset() {
        Charset detected = Charset.forName("ASCII");
        Charset result = charsetSupersets.decodeAs(detected);
        assertEquals(detected, result);
    }
}
